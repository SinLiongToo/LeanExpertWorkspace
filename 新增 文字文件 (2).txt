<script>

const categories=['Process','Material','Method','Machine'];
let chart;

// Tabs
function showTab(e,id){
document.querySelectorAll('.content').forEach(c=>c.classList.remove('active'));
document.querySelectorAll('.tab').forEach(t=>t.classList.remove('active'));
document.getElementById(id).classList.add('active');
e.currentTarget.classList.add('active');
}

////////////////////////////////////////////////
// ✅ 3×5 WHY (MULTI-CHAIN)
////////////////////////////////////////////////
let chainCount=0;

function addWhyChain(){

chainCount++;

const block=document.createElement('div');
block.className='card';

block.innerHTML=`
<b>Chain ${chainCount}</b>
<input placeholder="Why 1">
<input placeholder="Why 2">
<input placeholder="Why 3">
<input placeholder="Why 4">
<input placeholder="Why 5">
`;

document.getElementById('whyContainer').appendChild(block);
}

////////////////////////////////////////////////
// ✅ FISHBONE + NUMBER
////////////////////////////////////////////////
function initFish(){

const container=document.getElementById('fishContainer');

categories.forEach(cat=>{

   const div=document.createElement('div');
   div.className='bone';

   div.innerHTML=`
   <h4>${cat}</h4>
   <div class="items"></div>
   <button class="addBtn" onclick="addItem(this)">+ Add Cause</button>
   `;

   container.appendChild(div);
});
}

function addItem(btn){

const area=btn.parentElement.querySelector('.items');
const count=area.children.length+1;

const input=document.createElement('input');
input.placeholder='Cause '+count;

area.appendChild(input);
}

////////////////////////////////////////////////
// ✅ CHART
////////////////////////////////////////////////
function initChart(){
chart=new Chart(document.getElementById('chart'),{
   type:'bar',
   data:{
 	labels:categories,
 	datasets:[{
   	label:'Cause Count',
   	data:[0,0,0,0],
   	backgroundColor:['#ef4444','#f59e0b','#10b981','#3b82f6']
 	}]
   }
});
}

function updateChart(){

const counts=[...document.querySelectorAll('.bone')]
.map(b=>b.querySelectorAll('input').length);

chart.data.datasets[0].data=counts;
chart.update();
}

////////////////////////////////////////////////
// ✅ EXPORT
////////////////////////////////////////////////
function exportAll(){

const whyChains=[...document.querySelectorAll('#whyContainer .card')]
.map(c=>[...c.querySelectorAll('input')].map(i=>i.value));

const fishboneData=categories.reduce((acc,cat,i)=>{
   acc[cat]=[...document.querySelectorAll('.bone')[i]
   .querySelectorAll('input')].map(i=>i.value);
   return acc;
},{});

const data={
   d8:{
 	d1:d1.value,d2:d2.value,d3:d3.value,d4:d4.value,
 	d5:d5.value,d6:d6.value,d7:d7.value,d8:d8f.value
   },
   why:whyChains,
   fishbone:fishboneData
};

const blob=new Blob([JSON.stringify(data,null,2)]);
const a=document.createElement('a');
a.href=URL.createObjectURL(blob);
a.download='8d.json';
a.click();
}

////////////////////////////////////////////////
// ✅ ✅ LOAD MOCK (FULLY WORKING)
////////////////////////////////////////////////
function loadMock(){

// ✅ 8D
d1.value="Team: Test / Process / Supplier";
d2.value="Final test failure increased to 8%";
d3.value="Containment: Stop shipment + sorting";
d4.value="Solder + capacitor variation";
d5.value="Adjust process + supplier control";
d6.value="Failure reduced to 1.5%";
d7.value="Update SOP + SPC";
d8f.value="Team recognition";

// ✅ WHY RESET
const whyContainer=document.getElementById('whyContainer');
whyContainer.innerHTML="";
chainCount=0;

const chains=[
  ["Output unstable","Solder issue","Temp variation","Oven drift","No calibration"],
  ["Output unstable","Cap drift","Supplier change","No notice","Weak control"],
  ["Escape defect","Inspection miss","Low sampling","Old spec","No update"]
];

chains.forEach(arr=>{
   addWhyChain();
   const last=whyContainer.lastChild;
   last.querySelectorAll('input').forEach((inp,i)=>inp.value=arr[i]);
});

// ✅ FISHBONE RESET
const fishContainer=document.getElementById('fishContainer');
fishContainer.innerHTML="";
initFish();

const fishData={
   Process:["Temp fluctuation","Reflow unstable","Oven not calibrated"],
   Material:["Cap drift","Supplier change","Batch variation"],
   Method:["Old SOP","Low sampling","No update control"],
   Machine:["Sensor drift","Tester aging"]
};

document.querySelectorAll('.bone').forEach((bone,i)=>{
   const cat=categories[i];

   fishData[cat].forEach(val=>{
 	addItem(bone.querySelector('button'));
 	const inputs=bone.querySelectorAll('input');
 	inputs[inputs.length-1].value=val;
   });
});

// ✅ UPDATE CHART
updateChart();
}

////////////////////////////////////////////////
